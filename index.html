  return choose(s('scry')+' '+nm(top),[{label:s('top'),val:0},{label:s('bottom'),val:1}],v=>{if(v)deck.unshift(deck.pop());done();},turn,()=>P[turn].field.includes(top)?1:0);}

 if(cardPlayed==='R'){if(!opp.field.length){log(s('noTarget'));return done();}

  return choose(s('pickLand'),opp.field.map((c,i)=>({label:nm(c),val:i})),i=>{const c=opp.field.splice(i,1)[0];opp.grave.push(c);log(nm(c)+' '+s('destroyed'));sfx('destroy');done();},turn,o=>{const f=opp.field,n=k=>f.filter(x=>x===f[k.val]).length;let b=o[0];o.forEach(k=>{if(n(k)>n(b))b=k;});return b.val;});}

 if(cardPlayed==='B'){if(!opp.hand.length){log(s('noCards'));return done();}

  const idxs=shuffle(opp.hand.map((c,i)=>i)).slice(0,3);

  return choose(s('swamp'),idxs.map(i=>({label:nm(opp.hand[i]),val:i})),i=>{const c=opp.hand.splice(i,1)[0];opp.grave.push(c);log(gn(P.indexOf(opp))+' '+s('discarded')+' '+nm(c));sfx('discard');done();},turn,o=>{const x=o.find(k=>!opp.field.includes(opp.hand[k.val]));return (x||o[0]).val;});}

 if(cardPlayed==='G'){const types=[...new Set(me.grave)];if(!types.length){log(s('noCards'));return done();}

  return choose(s('forest'),types.map(c=>({label:nm(c),val:c})),c=>{me.grave.splice(me.grave.lastIndexOf(c),1);me.hand.push(c);log(gn(turn)+' '+s('back')+' '+nm(c));sfx('recover');done();},turn,o=>{const x=o.find(k=>!me.field.includes(k.val));return (x||o[0]).val;});}

 done();

}


function checkWin(){

 const f=P[turn].field,cnt={};f.forEach(c=>cnt[c]=(cnt[c]||0)+1);

 let msg=null;

 // Controlla Domain (5 terre di tipo diverso)

 const requiredTypes=['R','B','G','U','W'];

 const hasDomain=requiredTypes.every(t=>cnt[t]>=1);

 

 if(hasDomain) msg=s('winD');

 else for(const k in cnt)if(cnt[k]>=5)msg=s('winS')+' '+nm(k);

 

 if(msg){

  over=true;sfx(turn===0?'win':'lose');

  choose(gn(turn)+' '+msg,[{label:s('again'),val:1}],initDeck);

 }

}

 if(over||$('choiceModal').style.display==='flex')return;

 turn=1-turn;dropped=false;log('--- '+gn(turn)+' ---');sfx('pass');draw(turn);render();

 if(turn===1)aiLater(aiTurn,900);

}


function aiTurn(){

 if(over||turn!==1)return;const h=P[1].hand,f=P[1].field;

 if(!h.length)return aiLater(()=>endTurn(true),700);

 const d=cfg.diff;if(d==='nabbo'){playCard(Math.floor(Math.random()*h.length),true);return;}

 let best=-1,bi=0;

 h.forEach((c,i)=>{let sc=(d==='pro'?Math.random()*.3:Math.random())+f.filter(x=>x===c).length*.8;if(!f.includes(c))sc+=2;if(c==='R'&&P[0].field.length)sc+=1;if(c==='B'&&P[0].hand.length)sc+=.5;if(d==='pro'&&c==='R'&&new Set(P[0].field).size>=3)sc+=3;if(c==='G'&&P[1].grave.length)sc+=.5;if(c==='W'&&deck.length)sc+=.3;if(sc>best){best=sc;bi=i;}});

 playCard(bi,true);

}
