<script>

const colors={R:{name:'Montagna',cls:'mountain',eff:'ROSSO: Spacca 1 terra avversario'},B:{name:'Palude',cls:'swamp',eff:'NERO: Avv. scarta 1'},G:{name:'Foresta',cls:'forest',eff:'VERDE: Recupera dal cimitero'},U:{name:'Isola',cls:'island',eff:'BLU: Pesca 1 / FOIL COUNTER'},W:{name:'Pianura',cls:'plains',eff:'BIANCO: Ricopia terra tua'},C:{name:'Waste',cls:'colorless',eff:'JOLLY'}}

let deck=[], p=[{hand:[],field:[],grave:[]},{hand:[],field:[],grave:[]}], turn=0, counterPending=null;

function initDeck(){deck=[];['R','B','G','U','W'].forEach(c=>{for(let i=0;i<7;i++)deck.push(c)});for(let i=0;i<5;i++)deck.push('C');deck.sort(()=>Math.random()-.5); p.forEach(pl=>{pl.hand=[];pl.field=[];pl.grave=[]; for(let i=0;i<3;i++)pl.hand.push(deck.pop())}); update()}

function draw(){if(counterPending)return; if(deck.length==0)return log('Mazzo finito!'); p[turn].hand.push(deck.pop()); log(`G${turn+1} pesca`); checkCounterOpportunity(); update()}

function play(idx){let card=p[turn].hand[idx]; if(counterPending)return; if(counterPending!==null)return;

 // FOIL check avversario

 let opp=1-turn;

 if(p[opp].hand.some(c=>c=='U') && p[opp].hand.length>=2){

   if(confirm(`G${opp+1} vuoi fare FOIL COUNTER? Scarti Isola + 1 carta per annullare?`)){

     let uIdx=p[opp].hand.indexOf('U'); p[opp].hand.splice(uIdx,1); p[opp].grave.push('U'); p[opp].hand.splice(0,1); p[opp].grave.push('C');

     p[turn].grave.push(card); p[turn].hand.splice(idx,1); log(`G${opp+1} COUNTERA con FOIL!`); update(); return;

   }

 }

 p[turn].hand.splice(idx,1); p[turn].field.push(card); apply(card); log(`G${turn+1} gioca ${colors[card].name} - ${colors[card].eff}`); checkWin(); update();

}

function apply(c){

 if(c=='R'){let opp=1-turn; if(p[opp].field.length==0)return; let target=prompt(`Scegli indice terra avversaria da spaccare 0-${p[opp].field.length-1}: ${p[opp].field.join(',')}`); let i=parseInt(target); if(!isNaN(i)&&p[opp].field[i]){let destroyed=p[opp].field.splice(i,1)[0]; p[opp].grave.push(destroyed)}}

 if(c=='B'){let opp=1-turn; if(p[opp].hand.length==0)return; alert(`Mano avv: ${p[opp].hand.map(x=>colors[x].name).join(', ')}`); p[opp].hand.splice(0,1); p[opp].grave.push('C')}

 if(c=='G'){if(p[turn].grave.length==0)return; let rec=p[turn].grave.pop(); p[turn].hand.push(rec)}

 if(c=='U'){p[turn].hand.push(deck.pop()||'C')}

 if(c=='W'){if(p[turn].field.length<=1)return; let copy=p[turn].field[p[turn].field.length-2]; if(copy)apply(copy)}

}

function checkCounterOpportunity(){}

function checkWin(){let f=p[turn].field; if(f.length>=5){let uniq=[...new Set(f.filter(x=>x!='C'))]; if(uniq.length>=5){alert(`G${turn+1} VINCE con 5 COLORI DIVERSI!`); initDeck(); return} let counts={}; f.forEach(x=>{if(x!='C')counts[x]=(counts[x]||0)+1}); for(let k in counts) if(counts[k]>=5){alert(`G${turn+1} VINCE con 5 ${colors[k].name}!`); initDeck(); return}}}

function endTurn(){turn=1-turn; document.getElementById('turn').textContent='G'+(turn+1); log(`--- Turno G${turn+1} ---`); update()}

function log(t){let l=document.getElementById('log'); l.innerHTML=`> ${t}<br>`+l.innerHTML}

function update(){document.getElementById('deckCount').textContent=deck.length; document.getElementById('discardCount').textContent=p[0].grave.length+p[1].grave.length;

 let html=''; p.forEach((pl,i)=>{html+=`<div class="bg-zinc-900 p-3 rounded-xl ${i==turn?'ring-2 ring-yellow-400':''}"><h2 class="font-bold ${i==0?'text-green-400':'text-blue-400'}">GIOCATORE ${i+1} ${i==turn?'(TURNO)':''}</h2><div class="text-[10px] mt-1">CAMPO [${pl.field.length}]:</div><div class="flex flex-wrap gap-1 min-h-[56px]">`+pl.field.map(c=>`<div class="card ${colors[c].cls} w-12 h-16 rounded flex items-center justify-center text-[9px] font-bold">${colors[c].name[0]}</div>`).join('')+`</div><div class="text-[10px] mt-2">MANO:</div><div class="flex flex-wrap gap-1">`+pl.hand.map((c,idx)=>`<div onclick="${i==turn?`play(${idx})`:''}" class="card ${colors[c].cls} w-12 h-16 rounded flex flex-col items-center justify-center text-[8px] font-bold cursor-pointer border ${i!=turn?'opacity-50':''}"><span>${colors[c].name}</span><span class="text-[6px] mt-1">${colors[c].eff.slice(0,12)}</span></div>`).join('')+`</div><div class="text-[10px] mt-2">Cimitero: ${pl.grave.length}</div></div>`}); document.getElementById('boards').innerHTML=html}

initDeck();

</script>
